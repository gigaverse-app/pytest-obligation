def resolve_stalled(self, row_id: int) -> str:
    row = self.rows[row_id]
    if row.open_attempt is not None:
        row.state = "UNKNOWN"
        return "MARK_UNKNOWN"
    row.state = "READY"
    return "REARM_READY"
