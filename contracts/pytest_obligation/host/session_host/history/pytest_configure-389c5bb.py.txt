# Frozen evidence: pytest_plugin.pytest_configure exactly as shipped at
# 389c5bb84f7ab186fa9c7815c2e15cb9332b67a4, before #209 fixed it. The planted bug
# `shipped-389c5bb` runs this text in the real pytest_plugin module's namespace (plus
# the `configure` it imported then) and installs it as the plugin's hook, so the
# contract replays the code that leaked one session's host into the next.
# adapter.py checks its SHA-256; never edit it.

def pytest_configure(config: pytest.Config) -> None:
    from pytest_obligation import _cost_plugin, _debt_plugin
    from pytest_obligation._collection_coverage import CollectionCoverage

    config.pluginmanager.register(_debt_plugin, "due-work-debt")
    config.pluginmanager.register(_cost_plugin, "obligation-costs")
    if config.getoption("obligation_cost_baseline_complete") and not hasattr(config, "workerinput"):
        config.pluginmanager.register(_cost_plugin.CostCompleteness(config), "obligation-cost-completeness")
    config.pluginmanager.register(CollectionCoverage(config), "obligation-required-cases")
    if config.getoption("due_work_check_fixtures"):
        config.pluginmanager.register(
            import_module("pytest_obligation._internal.fixture_resolution"), "due-work-fixtures"
        )
    config.addinivalue_line(
        "markers", "due_work: a case generated by due-work-harness (contract, safety, exemption and selection suites)"
    )
    config.addinivalue_line("markers", "due_work_exploration: optional generated schedule search")
    if config.getoption("due_work_profile_report") or config.getoption(SUMMARY):
        config.pluginmanager.register(import_module("pytest_obligation.profiles.reporting"), "due-work-profile-report")
    if config.getoption("due_work_explore") != "off":
        try:
            import_module("hypothesis")
        except ModuleNotFoundError as error:
            raise pytest.UsageError("Install pytest-obligation[exploration] to use --due-work-explore") from error
    # Under xdist, the controller receives every worker's reports and holds the verdict; a worker
    # sees only its share, and would rescan the whole project at its own session's end for nothing.
    worker = hasattr(config, "workerinput")
    if config.getoption(VERIFY) and not worker:
        config.pluginmanager.register(_SuitesRan(config), VERIFY_PLUGIN)
    if config.getoption(SUMMARY) and not worker:
        config.pluginmanager.register(_Summary(), SUMMARY_PLUGIN)
    if config.getoption(RECORD):
        # Histories run in the process that owns the database; under xdist that is a worker, whose
        # recorder the controller never sees, so recording would silently print nothing.
        if worker or config.getoption("numprocesses", default=None):
            raise pytest.UsageError(f"{RECORD} records in the process that runs the histories: run without -n")
        config.pluginmanager.register(_Recording(), RECORD_PLUGIN)
    path = config.getini("due_work_harness_host")
    if not path:
        return
    module_name, _, attribute = path.partition(":")
    assert attribute, f"due_work_harness_host must look like 'package.module:attribute', got {path!r}"

    def configured_host() -> Host:
        target = getattr(import_module(module_name), attribute)
        host = target if isinstance(target, Host) else target()
        assert isinstance(host, Host), f"{path} did not produce a pytest_obligation Host"
        return host

    configure(configured_host)
